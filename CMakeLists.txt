cmake_minimum_required(VERSION 3.22)
project(TrailerForce VERSION 0.4.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  # Ship a self-contained VST3 without requiring a separate VC++ runtime installer.
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
option(TF_CORE_ONLY "Build portable MIDI engine tests without JUCE" OFF)
add_library(tf_core Source/Core/Engine.cpp Source/Core/Profiles.cpp Source/Core/Generative.cpp Source/Core/MidiFile.cpp Source/Core/Sound.cpp)
target_include_directories(tf_core PUBLIC Source/Core)
set_target_properties(tf_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
include(CTest)
if(BUILD_TESTING)
  add_executable(tf_tests tests/CoreTests.cpp)
  target_link_libraries(tf_tests PRIVATE tf_core)
  add_test(NAME engine COMMAND tf_tests)
  find_package(Python3 COMPONENTS Interpreter QUIET)
  if(Python3_Interpreter_FOUND)
    add_test(NAME midi_integrity COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/VerifyMidi.py ${CMAKE_CURRENT_BINARY_DIR}/test-arrangement.mid)
    add_test(NAME midi_expression COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/VerifyMidi.py ${CMAKE_CURRENT_BINARY_DIR}/test-expression.mid)
    set_tests_properties(midi_integrity midi_expression PROPERTIES DEPENDS engine)
  endif()
endif()
if(TF_CORE_ONLY)
  return()
endif()
set(JUCE_SOURCE_DIR "" CACHE PATH "Existing JUCE checkout (optional)")
if(JUCE_SOURCE_DIR)
  add_subdirectory("${JUCE_SOURCE_DIR}" JUCE)
else()
  include(FetchContent)
  FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.6 GIT_SHALLOW TRUE)
  FetchContent_MakeAvailable(JUCE)
endif()
function(tf_plugin target product code midi_only)
  if(midi_only)
    set(formats AU)
    set(synth FALSE)
  else()
    set(formats VST3)
    if(APPLE)
      list(APPEND formats AU)
    endif()
    set(synth TRUE)
  endif()
  juce_add_plugin(${target}
    COMPANY_NAME "Vinci Sounds" BUNDLE_ID "com.vincisounds.${target}"
    PLUGIN_MANUFACTURER_CODE Vnci PLUGIN_CODE ${code}
    PRODUCT_NAME "${product}" FORMATS ${formats}
    IS_SYNTH ${synth} IS_MIDI_EFFECT ${midi_only}
    NEEDS_MIDI_INPUT TRUE NEEDS_MIDI_OUTPUT TRUE
    EDITOR_WANTS_KEYBOARD_FOCUS TRUE COPY_PLUGIN_AFTER_BUILD FALSE
    VST3_CATEGORIES Instrument Generator)
  target_sources(${target} PRIVATE Source/Plugin/Processor.cpp Source/Plugin/Editor.cpp)
  target_compile_definitions(${target} PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
  target_link_libraries(${target} PRIVATE tf_core juce::juce_audio_utils PUBLIC juce::juce_recommended_config_flags)
endfunction()
tf_plugin(TrailerForce "Trailer Force" TrFc FALSE)
if(APPLE)
  tf_plugin(TrailerForceMidi "Trailer Force MIDI FX" TrFm TRUE)
endif()
